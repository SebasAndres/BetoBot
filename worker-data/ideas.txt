proc registrarPartidoEnArbol(m):
    if (m.t1 in tree.level_1) and (m.t2 in tree.level_1[m.t1]):
        updateHSET in tree.level_1[m.t1][m.t2]
        return

    elif (m.t2 in tree.level_1) and (m.t1 in tree.level_1[m.t2]):
        updateHSET in tree.level_1[m.t2][m.t1]

    elif (m.t1 in tree.level_1):
        tree.level_1[m.t1].agregarNuevoHSET(m.t2)
    
    elif (m.t2 in tree.level_1):
        tree.level_1[m.t2].agregarNuevoHSET(m.t1) 

    else:
        tree.level_1.agregarNuevoHSET(m.t1)
        tree.level_1[m.t1].agregarNuevoHSET(m.t2)
